function(lvd_gui_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE lvdgui_app Qt6::Test)
    lvd_target_warnings(${name})
    target_compile_definitions(${name} PRIVATE LVD_KERNEL_BUILD="${CMAKE_SOURCE_DIR}/kernel/nvc/build")
    add_test(NAME ${name} COMMAND ${name})

    # The MinGW Qt DLLs must be on PATH when the test runs.
    get_target_property(_qt_core_loc Qt6::Core IMPORTED_LOCATION_RELEASE)
    if(NOT _qt_core_loc)
        get_target_property(_qt_core_loc Qt6::Core IMPORTED_LOCATION)
    endif()
    set(_env "QT_QPA_PLATFORM=offscreen")
    if(WIN32 AND _qt_core_loc)
        get_filename_component(_qt_bin "${_qt_core_loc}" DIRECTORY)
        list(APPEND _env "PATH=${_qt_bin};$ENV{PATH}")
    endif()
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "${_env}")
endfunction()
